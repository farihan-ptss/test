Tetapan Pautan Webhook Google Sheet (Apps Script)

Panduan Ringkas Integrasi Google Sheet:

  1. Buka Google Sheet baharu dan pergi ke Extensions > Apps Script.
  2. Padamkan kod asal dan gantikan dengan skrip di bawah:
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Tambah header jika helai masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Masa Hantar", "Nama Pelajar", "Markah Latihan", "Peratusan (%)"]);
    }
    
    sheet.appendRow([
      new Date(),
      data.nama,
      data.latihan + "/10",
      data.jumlahMarkah + "%"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
